# IPv6 Support Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Protect IPv6 destinations exactly like IPv4 ones in the XDP filter and the loader, without breaking existing IPv4 configs.

**Architecture:** One 16-byte address type (`struct addr6`, IPv4 stored v4-mapped `::ffff:a.b.c.d`) replaces every `__u32` address in map keys and values. A pure, map-free L3 parser (`xdp/l3parse.h`) turns IPv4 and IPv6 packets (with a bounded extension header walk) into a `struct l3` descriptor that all program parts consume instead of `struct iphdr`. The Rust loader gets one shared `Addr6` mirror type in `loader/addr.rs`, and every module switches from `Ipv4Addr`/`SocketAddrV4` to `IpAddr`/`SocketAddr`.

**Tech Stack:** C (eBPF, clang `-target bpf -mcpu=v3`), Rust 2024 + aya 0.14, Python 3 (e2e), docker (privileged e2e container).

**Spec:** `docs/superpowers/specs/2026-11-11-ipv6-design.md`

## Global Constraints

- Existing IPv4-only `config.toml` files must load unchanged and behave the same.
- IPv4 is stored v4-mapped: bytes `00×10, ff, ff, a, b, c, d`.
- LPM trie entries for IPv4 prefixes use `prefixlen + 96`.
- IPv6 source prefix for throttle, rate limit, strikes, bans and penalty: `[filter] ipv6_source_prefix`, default `64`, valid `32..=128`. IPv4 always per address (/32).
- Extension header walk: at most `MAX_EXT_HDRS = 4` headers (hop-by-hop 0, routing 43, fragment 44, destination options 60); more, unknown-after-limit, or truncated → unparseable.
- Unparseable or fragmented packets to a protected target → drop; to anything else → pass.
- ICMPv6 types 133–137 (router/neighbour discovery) are never dropped by any feature.
- Metric labels carry addresses in canonical text (Rust `Display`, RFC 5952); v4-mapped addresses are shown as plain IPv4.
- Minimum kernel stays 5.15; the program must pass the verifier.
- Every commit: Conventional Commits in English, author is the repository user only, no Claude trailers. Branch: `feat/ipv6` (already created).
- `cargo test`, `cargo clippy --all-targets -- -D warnings` and `cargo fmt --check` pass after every task.

```c
static size_t build_v4(__u8 *p, __u8 proto, __u16 frag_off, size_t payload)

static void test_parse_v4(void)
{
    __u8 p[128];
    struct l3 l;
    size_t n = build_v4(p, 6, 0, 20);
    CHECK(run_parse(p, n, &l) == L3_OK);
    CHECK(l.family == FAMILY_V4 && l.proto == 6 && l.l4off == 20 && l.l3len == 40 && l.hdrlen == 20);
    CHECK(l.frag == FRAG_NONE);
    CHECK(addr_is_v4(&l.src) && addr_v4(&l.src) == htonl(0x01020304));

    n = build_v4(p, 17, 0x2000, 8); // MF, offset 0
    CHECK(run_parse(p, n, &l) == L3_OK && l.frag == FRAG_FIRST);
    n = build_v4(p, 17, 0x0001, 8); // offset 8 bytes
    CHECK(run_parse(p, n, &l) == L3_OK && l.frag == FRAG_LATER);

    n = build_v4(p, 6, 0, 0);
    p[0] = 0x44; // ihl 4 < 5
    CHECK(run_parse(p, n, &l) == L3_BAD);
    CHECK(run_parse(p, 19, &l) == L3_BAD); // truncated header
    p[0] = 0x00;
    CHECK(run_parse(p, n, &l) == L3_NOT_IP);
}

static void test_parse_v6(void)
{
    __u8 p[256];
    struct l3 l;
    size_t n = build_v6(p, 6, 20);
    CHECK(run_parse(p, n, &l) == L3_OK);
    CHECK(l.family == FAMILY_V6 && l.proto == 6 && l.l4off == 40 && l.l3len == 60 && l.hdrlen == 40);
    CHECK(!addr_is_v4(&l.src) && l.src.w[0] == htonl(0x20010db8) && l.dst.w[3] == htonl(2));
    CHECK(l.frag == FRAG_NONE);

    // hop-by-hop (8 bytes) then destination options (16 bytes) then udp
    n = build_v6(p, 0, 8 + 16 + 8);
    p[40] = 60; p[41] = 0;
    p[48] = 17; p[49] = 1;
    CHECK(run_parse(p, n, &l) == L3_OK && l.proto == 17 && l.l4off == 64 && l.hdrlen == 64);

    // fragment header, offset 0 with M flag -> first fragment
    n = build_v6(p, 44, 8 + 8);
    p[40] = 17; p[42] = 0x00; p[43] = 0x01;
    CHECK(run_parse(p, n, &l) == L3_OK && l.frag == FRAG_FIRST && l.proto == 17 && l.l4off == 48);
    // offset 8 bytes -> later fragment
    p[42] = 0x00; p[43] = 0x08;
    CHECK(run_parse(p, n, &l) == L3_OK && l.frag == FRAG_LATER);
    // atomic fragment (offset 0, no M) is still a fragment header -> first
    p[43] = 0x00;
    CHECK(run_parse(p, n, &l) == L3_OK && l.frag == FRAG_FIRST);

    // five hop-by-hop headers: over the limit
    n = build_v6(p, 0, 5 * 8 + 8);
    for (int i = 0; i < 5; i++)
    {
        p[40 + i * 8] = i == 4 ? 17 : 0;
    }
    CHECK(run_parse(p, n, &l) == L3_BAD);

    // truncated extension header: family and addresses are still known
    n = build_v6(p, 0, 4);
    CHECK(run_parse(p, n, &l) == L3_BAD);
    CHECK(l.family == FAMILY_V6 && l.dst.w[3] == htonl(2));
    // truncated fixed header: family known, addresses zero
    CHECK(run_parse(p, 39, &l) == L3_BAD);
    CHECK(l.family == FAMILY_V6 && l.dst.w[0] == 0 && l.dst.w[3] == 0);

    // icmpv6 passes the parser with proto 58
    n = build_v6(p, 58, 8);
    p[40] = 135;
    CHECK(run_parse(p, n, &l) == L3_OK && l.proto == 58);
}
```

```c
/*
 * One address type for both families: IPv6 as is, IPv4 v4-mapped
 * (::ffff:a.b.c.d). All words in network byte order, so an IPv4 address
 * taken from a header lands in w[3] unchanged.
 */
struct addr6
{
    __u32 w[4];
};
_Static_assert(sizeof(struct addr6) == 16, "addr6 size mismatch!");

static __always_inline struct addr6 addr_from_v4(const __u32 be)
{
    struct addr6 a = {{0, 0, bpf_htonl(0x0000ffff), be}};
    return a;
}

static __always_inline __u8 addr_is_v4(const struct addr6 *a)
{
    return a->w[0] == 0 && a->w[1] == 0 && a->w[2] == bpf_htonl(0x0000ffff);
}

static __always_inline __u32 addr_v4(const struct addr6 *a)
{
    return a->w[3];
}

static __always_inline __u8 addr_eq(const struct addr6 *a, const struct addr6 *b)
{
    return a->w[0] == b->w[0] && a->w[1] == b->w[1] && a->w[2] == b->w[2] && a->w[3] == b->w[3];
}

// keeps the first `len` bits (0..128) of a
static __always_inline struct addr6 addr_mask(struct addr6 a, const __u32 len)
{
#pragma unroll
    for (__u32 i = 0; i < 4; i++)
    {
        const __s32 bits = (__s32)len - (__s32)(i * 32);
        if (bits <= 0)
        {
            a.w[i] = 0;
        }
        else if (bits < 32)
        {
            a.w[i] &= bpf_htonl(~0u << (32 - bits));
        }
    }
    return a;
}

// key identifying one flow (addresses and ports in network byte order)
struct flow_key
{
    struct addr6 src;
    struct addr6 dst;
    __u16 src_port;
    __u16 dst_port;
};
_Static_assert(sizeof(struct flow_key) == 36, "flow_key size mismatch!");

static __always_inline struct flow_key gen_flow_key(const struct addr6 *src, const struct addr6 *dst, const __u16 src_port, const __u16 dst_port)
{
    struct flow_key key = {.src = *src, .dst = *dst, .src_port = src_port, .dst_port = dst_port};
    return key;
}
```

- [ ] **Step 4: Create `xdp/l3parse.h`**

```c
#ifndef L3PARSE_H
#define L3PARSE_H

/*
 * Pure L3 parser shared by the XDP and tc programs (and the native unit
 * tests): no maps, no helpers besides byte order macros. `ip` points at the
 * first byte of the IPv4 or IPv6 header.
 */

#include <linux/types.h>
#include "common.h"

#define FAMILY_V4 4
#define FAMILY_V6 6

#define FRAG_NONE 0
#define FRAG_FIRST 1 // offset 0 (more fragments follow, or atomic fragment)
#define FRAG_LATER 2 // offset != 0, no l4 header

#define L3_OK 0
#define L3_NOT_IP 1 // not IPv4/IPv6: not our business
#define L3_BAD 2    // IP, but malformed or too many extension headers

#define MAX_EXT_HDRS 4
// extension headers end before this offset or the packet is L3_BAD; keeps
// every offset bounded for the verifier
#define MAX_L3_HDRLEN 512

#define NEXTHDR_HOP 0
#define NEXTHDR_ROUTING 43
#define NEXTHDR_FRAGMENT 44
#define NEXTHDR_DEST 60

struct l3
{
    struct addr6 src;
    struct addr6 dst;
    __u32 l4off;  // offset of the l4 header from `ip`
    __u32 l3len;  // IPv4 tot_len, IPv6 40 + payload length
    __u32 hdrlen; // all IP headers incl. extension headers (== l4off)
    __u8 family;  // FAMILY_V4 / FAMILY_V6
    __u8 proto;   // l4 protocol after all extension headers
    __u8 frag;    // FRAG_*
    __u8 tos;     // IPv4 tos / IPv6 traffic class
};

static __always_inline int parse_l3_v4(const __u8 *ip, const void *data_end, struct l3 *out)
{
    out->family = FAMILY_V4; // set first: callers read it on L3_BAD too
    if ((const void *)(ip + 20) > data_end)
    {
        return L3_BAD;
    }
    const __u32 ihl = (ip[0] & 0x0f) * 4;
    if (ihl < 20 || (const void *)(ip + ihl) > data_end)
    {
        return L3_BAD;
    }
    __u16 tot, frag_off;
    __builtin_memcpy(&tot, ip + 2, 2);
    __builtin_memcpy(&frag_off, ip + 6, 2);
    __u32 s, d;
    __builtin_memcpy(&s, ip + 12, 4);
    __builtin_memcpy(&d, ip + 16, 4);
    out->src = addr_from_v4(s);
    out->dst = addr_from_v4(d);
    out->family = FAMILY_V4;
    out->proto = ip[9];
    out->tos = ip[1];
    out->l4off = ihl;
    out->hdrlen = ihl;
    out->l3len = bpf_ntohs(tot);
    frag_off = bpf_ntohs(frag_off);
    out->frag = (frag_off & 0x1fff) ? FRAG_LATER : (frag_off & 0x2000) ? FRAG_FIRST : FRAG_NONE;
    return L3_OK;
}

static __always_inline int parse_l3_v6(const __u8 *ip, const void *data_end, struct l3 *out)
{
    out->family = FAMILY_V6; // set first: callers read it on L3_BAD too
    if ((const void *)(ip + 40) > data_end)
    {
        return L3_BAD;
    }
    __u16 plen;
    __builtin_memcpy(&plen, ip + 4, 2);
    __builtin_memcpy(&out->src, ip + 8, 16);
    __builtin_memcpy(&out->dst, ip + 24, 16);
    out->family = FAMILY_V6;
    out->tos = ((ip[0] & 0x0f) << 4) | (ip[1] >> 4);
    out->l3len = 40 + bpf_ntohs(plen);
    out->frag = FRAG_NONE;

    __u8 next = ip[6];
    __u32 off = 40;
#pragma unroll
    for (__u32 i = 0; i < MAX_EXT_HDRS; i++)
    {
        if (next != NEXTHDR_HOP && next != NEXTHDR_ROUTING && next != NEXTHDR_FRAGMENT && next != NEXTHDR_DEST)
        {
            break;
        }
        const __u8 *h = ip + off;
        if ((const void *)(h + 8) > data_end)
        {
            return L3_BAD;
        }
        if (next == NEXTHDR_FRAGMENT)
        {
            __u16 fo;
            __builtin_memcpy(&fo, h + 2, 2);
            out->frag = (bpf_ntohs(fo) & 0xfff8) ? FRAG_LATER : FRAG_FIRST;
            off += 8;
        }
        else
        {
            off += ((__u32)h[1] + 1) * 8;
        }
        next = h[0];
        if (off > MAX_L3_HDRLEN)
        {
            return L3_BAD;
        }
    }
    if (next == NEXTHDR_HOP || next == NEXTHDR_ROUTING || next == NEXTHDR_FRAGMENT || next == NEXTHDR_DEST)
    {
        return L3_BAD; // more than MAX_EXT_HDRS extension headers
    }
    out->proto = next;
    out->l4off = off;
    out->hdrlen = off;
    return L3_OK;
}

// *out is always fully initialised (zeroed first), also on L3_BAD/L3_NOT_IP;
// a truncated IPv6 header leaves dst = ::
static __always_inline int parse_l3(const void *ip, const void *data_end, struct l3 *out)
{
    __builtin_memset(out, 0, sizeof(*out));
    const __u8 *p = ip;
    if ((const void *)(p + 1) > data_end)
    {
        return L3_NOT_IP;
    }
    switch (p[0] >> 4)
    {
    case 4:
        return parse_l3_v4(p, data_end, out);
    case 6:
        return parse_l3_v6(p, data_end, out);
    }
    return L3_NOT_IP;
}

#endif
```

```c
 * Interfaces without a link layer header (GRE, IPIP, WireGuard): their

 * them here by index; L3_INTERFACES skips the lookup on routers that have

#include <linux/if_ether.h>
#include "l3parse.h"

struct

    __uint(max_entries, 64);
    __type(key, __u32);  // ifindex
    __type(value, __u8); // 1

/*
 * Parses the IP header of a packet on `ifindex` into *out. Returns the
 * L3_* result; *ip_out points at the IP header on L3_OK and L3_BAD.
 */
static __always_inline int packet_l3(const void *data, const void *data_end, __u32 ifindex,
                                     struct l3 *out, const void **ip_out)
{
    const void *ip = data;
    if (!(L3_INTERFACES && bpf_map_lookup_elem(&l3_interfaces, &ifindex)))
    {
        const struct ethhdr *eth = data;
        if ((const void *)(eth + 1) > data_end)
        {
            return L3_NOT_IP;
        }
        if (eth->h_proto != bpf_htons(ETH_P_IP) && eth->h_proto != bpf_htons(ETH_P_IPV6))
        {
            return L3_NOT_IP;
        }
        ip = eth + 1;
    }
    *ip_out = ip;
    return parse_l3(ip, data_end, out);
}

// the source as counted by throttle, rate limit, strikes, bans and penalty:
// IPv4 per address, IPv6 per IPV6_SOURCE_PREFIX
static __always_inline struct addr6 src_prefix(const struct addr6 *src)
{
    if (addr_is_v4(src))
    {
        return *src;
    }
    return addr_mask(*src, IPV6_SOURCE_PREFIX);
}

// advertisement and redirect: never dropped, or the host loses its neighbours

    if (l->family != FAMILY_V6 || l->proto != 58)
    {
        return 0;

    if ((const void *)(icmp + 1) > data_end)
    {
        return 0;
    }
    return icmp[0] >= 133 && icmp[0] <= 137;
}
```

```c
        return XDP_PASS;
    }

    {

    }
```

```c
    {
        return XDP_PASS;
    }

    if (l.frag == FRAG_FIRST)
    {
        return XDP_DROP;
    }
```

```rust
//! One address type for both families, mirroring `struct addr6`
//! (xdp/common.h): IPv6 as is, IPv4 v4-mapped (::ffff:a.b.c.d), bytes in
//! network order.

use std::fmt;
use std::net::{IpAddr, Ipv6Addr};

use aya::Pod;
use aya::maps::lpm_trie::Key;

#[repr(C, align(4))]
#[derive(Copy, Clone, Debug, Default, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct Addr6(pub [u8; 16]);

// SAFETY: repr(C) wrapper of a byte array, valid for any bit pattern.
unsafe impl Pod for Addr6 {}

impl Addr6 {
    pub fn from_ip(ip: IpAddr) -> Self {
        match ip {
            IpAddr::V4(v4) => Self(v4.to_ipv6_mapped().octets()),
            IpAddr::V6(v6) => Self(v6.octets()),
        }
    }

    pub fn to_ip(&self) -> IpAddr {
        let v6 = Ipv6Addr::from(self.0);
        match v6.to_ipv4_mapped() {
            Some(v4) => IpAddr::V4(v4),
            None => IpAddr::V6(v6),
        }
    }

    pub fn is_v4(&self) -> bool {
        matches!(self.to_ip(), IpAddr::V4(_))
    }

    /// Keeps the first `len` bits (0..=128) of the 128-bit form.
    pub fn mask(self, len: u8) -> Self {
        let bits = u128::from_be_bytes(self.0);
        let mask = if len == 0 { 0 } else { u128::MAX << (128 - u32::from(len.min(128))) };
        Self((bits & mask).to_be_bytes())
    }
}

impl fmt::Display for Addr6 {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        self.to_ip().fmt(f)
    }
}

/// A network of either family; `len` is in the family's own bits.
#[derive(Copy, Clone, Debug, PartialEq, Eq, Hash)]
pub struct Net {
    pub addr: IpAddr,
    pub len: u8,
}

impl Net {
    /// Prefix length in the 128-bit (v4-mapped) form.
    pub fn lpm_len(&self) -> u32 {
        match self.addr {
            IpAddr::V4(_) => u32::from(self.len) + 96,
            IpAddr::V6(_) => u32::from(self.len),
        }
    }

    /// The network address, masked, in the 128-bit form.
    pub fn addr6(&self) -> Addr6 {
        Addr6::from_ip(self.addr).mask(self.lpm_len() as u8)
    }

    /// The netmask in the 128-bit form.
    pub fn mask6(&self) -> Addr6 {
        Addr6([0xff; 16]).mask(self.lpm_len() as u8)
    }

    pub fn contains(&self, ip: IpAddr) -> bool {
        if ip.is_ipv4() != self.addr.is_ipv4() {
            return false;
        }
        Addr6::from_ip(ip).mask(self.lpm_len() as u8) == self.addr6()
    }
}

impl fmt::Display for Net {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(f, "{}/{}", self.addr6(), self.len)
    }
}

/// LPM trie key of `struct net_key` (xdp/reflection.h).
pub fn lpm_key(net: &Net) -> Key<Addr6> {
    Key::new(net.lpm_len(), net.addr6())
}

/// The source as counted by throttle, rate limit and bans (see src_prefix()
/// in xdp/l3.h): IPv4 per address, IPv6 per `v6_len`.
pub fn source_prefix(ip: IpAddr, v6_len: u8) -> Net {
    let len = if ip.is_ipv4() { 32 } else { v6_len };
    let net = Net { addr: ip, len };
    Net { addr: net.addr6().to_ip(), len }
}
```

```rust
/// Parses "ip:port/tcp" or "[ipv6]:port/udp" into the address and its IP protocol.
pub fn parse_exclude(entry: &str) -> Result<(std::net::SocketAddr, u8)> {
    let (addr, proto) = entry.rsplit_once('/').with_context(|| {
        format!("expected \"ip:port/tcp\" or \"[ipv6]:port/udp\", got '{entry}'")
    })?;
    let proto = match proto {
        "tcp" => libc::IPPROTO_TCP as u8,
        "udp" => libc::IPPROTO_UDP as u8,
        other => bail!("unknown protocol '{other}', expected tcp or udp"),
    };
    Ok((parse_socket_addr(addr)?, proto))
}

/// Parses "a.b.c.d/len" or "ipv6/len"; without a length the network is a
/// single address.
pub fn parse_cidr(entry: &str) -> Result<crate::addr::Net> {
    let (addr, len) = match entry.split_once('/') {
        Some((addr, len)) => (addr, Some(len)),
        None => (entry, None),
    };
    let addr: std::net::IpAddr = addr
        .parse()
        .with_context(|| format!("invalid network address in '{entry}'"))?;
    let max = if addr.is_ipv4() { 32 } else { 128 };
    let len: u8 = match len {
        Some(len) => len.parse().with_context(|| format!("invalid prefix length in '{entry}'"))?,
        None => max,
    };
    if len > max {
        bail!("prefix length of '{entry}' must be <= {max}");
    }
    Ok(crate::addr::Net { addr, len })
}
```

```python
NEXTHDR_HOP, NEXTHDR_FRAGMENT, NEXTHDR_DEST, NEXTHDR_UDP = 0, 44, 60, 17


def _checksum(data):
    if len(data) % 2:
        data += b"\x00"
    total = sum(struct.unpack("!%dH" % (len(data) // 2), data))
    while total >> 16:
        total = (total & 0xFFFF) + (total >> 16)
    return ~total & 0xFFFF


def _v6_raknet_ping(host, port, ext_types):
    """Sends a RakNet unconnected ping to host:port behind the given IPv6
    extension headers (raw socket, header included) and returns True when a
    pong comes back within 2 s."""
    src = os.environ.get("E2E_BIND", "fd99:0::2")
    payload = b"\x01" + struct.pack(">q", 1) + MAGIC + struct.pack(">q", 2)
    rx = socket.socket(socket.AF_INET6, socket.SOCK_DGRAM)
    rx.bind((src, 0))
    rx.settimeout(2)
    sport = rx.getsockname()[1]

    s6 = socket.inet_pton(socket.AF_INET6, src)
    d6 = socket.inet_pton(socket.AF_INET6, host)
    udp_len = 8 + len(payload)
    udp = struct.pack("!HHHH", sport, port, udp_len, 0) + payload
    pseudo = s6 + d6 + struct.pack("!I3xB", udp_len, NEXTHDR_UDP)
    udp = udp[:6] + struct.pack("!H", _checksum(pseudo + udp)) + udp[8:]

    chain = list(ext_types) + [NEXTHDR_UDP]
    exts = b""
    for i, kind in enumerate(ext_types):
        nxt = chain[i + 1]
        if kind == NEXTHDR_FRAGMENT:
            # offset 0, M flag clear: an atomic fragment (RFC 6946), which
            # linux delivers like an unfragmented packet
            exts += struct.pack("!BBHI", nxt, 0, 0, 0x12345678)
        else:
            # 8 bytes: next header, length 0, PadN option with 4 zero bytes
            exts += struct.pack("!BB", nxt, 0) + b"\x01\x04\x00\x00\x00\x00"
    ip6 = struct.pack("!IHBB", 6 << 28, len(exts) + udp_len, chain[0], 64) + s6 + d6

    raw = socket.socket(socket.AF_INET6, socket.SOCK_RAW, socket.IPPROTO_RAW)
    raw.sendto(ip6 + exts + udp, (host, 0))
    try:
        data, _ = rx.recvfrom(2048)
        return data[:1] == b"\x1c"  # unconnected pong
    except socket.timeout:
        return False
    finally:
        raw.close()
        rx.close()


def v6_ext_chain(host, port, count):
    """count extension headers: one hop-by-hop first (it must come first),
    then destination options."""
    count = int(count)
    kinds = ([NEXTHDR_HOP] if count else []) + [NEXTHDR_DEST] * max(count - 1, 0)
    return _v6_raknet_ping(host, port, kinds)


def v6_fragment(host, port):
    return _v6_raknet_ping(host, port, [NEXTHDR_FRAGMENT])
```
